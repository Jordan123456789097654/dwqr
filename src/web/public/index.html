Universal Roblox Product Whitelist Script

Paste this script into ServerScriptService in any Roblox Studio place to instantly protect all your products with the Axis Core Retail Whitelist API.

--!strict -- Axis Core Retail — Universal Product Whitelist Script -- Place this Script inside ServerScriptService in Roblox Studio local HttpService = game:GetService("HttpService") local Players = game:GetService("Players") local API_URL = "http://localhost:3000" -- Replace with your hosted backend domain local API_SECRET = "your_secure_api_secret_here" -- Must match ROBLOX_API_SECRET in .env local PRODUCT_ID = "vip_pass" -- Change for each product local function verifyPlayerWhitelist(player: Player): boolean local url = API_URL .. "/api/v1/roblox/verify" local payload = HttpService:JSONEncode({ robloxUserId = tostring(player.UserId), productId = PRODUCT_ID, placeId = game.PlaceId }) local ok, response = pcall(function() return HttpService:RequestAsync({ Url = url, Method = "POST", Headers = { ["Content-Type"] = "application/json", ["x-api-secret"] = API_SECRET }, Body = payload }) end) if ok and response and response.StatusCode == 200 then local data = HttpService:JSONDecode(response.Body) return data.success and data.whitelisted == true end warn("[Axis Core Protection] API Whitelist verification failed for " .. player.Name) return false end -- Check player on join Players.PlayerAdded:Connect(function(player) local isWhitelisted = verifyPlayerWhitelist(player) if not isWhitelisted then warn("[Axis Core Security] Kicking non-whitelisted player: " .. player.Name) player:Kick("❌ Access Denied: You do not own a valid Axis Core Retail whitelist for " .. PRODUCT_ID) else print("[Axis Core Security] Whitelist verified for " .. player.Name) end end)
